// Real libmpv/FFmpeg exercise, outside product code. Headless Windows software
// decoding is not an iPhone, GPU-rendering, HDR, power or thermal acceptance test.
import 'dart:async';
import 'dart:convert';
import 'dart:io';

import 'package:media_kit/media_kit.dart';
import 'package:cinema_client/player/native_player.dart';

Future<void> until(bool Function() predicate) async {
  final limit = DateTime.now().add(const Duration(seconds: 15));
  while (!predicate()) {
    if (DateTime.now().isAfter(limit)) throw StateError('Native engine event timed out');
    await Future<void>.delayed(const Duration(milliseconds: 30));
  }
}

void check(bool condition, String message) { if (!condition) throw StateError(message); }

Future<void> main(List<String> args) async {
  final library = args[0];
  final sample = File(args[1]);
  final subtitle = args[2];
  MediaKit.ensureInitialized(libmpv: library);
  final server = await HttpServer.bind(InternetAddress.loopbackIPv4, 0);
  final ranges = <String>[];
  var authorized = 0;
  server.listen((request) async {
    if (request.headers.value('x-cinema-test') != 'synthetic') {
      request.response.statusCode = 403;
      await request.response.close();
      return;
    }
    authorized++;
    final length = await sample.length();
    final range = request.headers.value('range');
    var start = 0;
    var end = length - 1;
    if (range != null) {
      ranges.add(range);
      final match = RegExp(r'^bytes=(\d+)-(\d*)$').firstMatch(range);
      if (match != null) {
        start = int.parse(match.group(1)!);
        if (match.group(2)!.isNotEmpty) end = int.parse(match.group(2)!);
        end = end.clamp(start, length - 1);
      }
    }
    if (start >= length) {
      request.response.statusCode = 416;
      await request.response.close();
      return;
    }
    request.response.headers.set('accept-ranges', 'bytes');
    request.response.headers.contentType = ContentType('video', 'x-matroska');
    request.response.contentLength = end - start + 1;
    if (range != null) {
      request.response.statusCode = 206;
      request.response.headers.set('content-range', 'bytes $start-$end/$length');
    }
    if (request.method == 'HEAD') { await request.response.close(); return; }
    try { await sample.openRead(start, end + 1).pipe(request.response); }
    on IOException { /* Native seek/dispose can intentionally close a response. */ }
  });
  final player = createNativePlayer();
  final native = player.platform as NativePlayer;
  final errors = <String>[];
  final errorSubscription = player.stream.error.listen(errors.add);
  try {
    await native.setProperty('ao', 'null');
    await native.setProperty('vid', 'auto');
    await native.setProperty('hwdec', 'no');
    await native.setProperty('cache', 'no');
    final media = 'http://127.0.0.1:${server.port}/synthetic.mkv';
    const headers = {'X-Cinema-Test': 'synthetic', 'User-Agent': 'cinema-native-regression'};
    await player.open(Media(media, httpHeaders: headers), play: false);
    await until(() => player.state.duration.inSeconds >= 7 && player.state.tracks.audio.where((t) => t.id != 'no' && t.id != 'auto').length == 2);
    final width = await native.getProperty('width');
    final height = await native.getProperty('height');
    final codec = await native.getProperty('video-codec');
    final pixels = await native.getProperty('video-dec-params/pixelformat');
    check(width == '3840' && height == '2160', 'Unexpected video dimensions: $width x $height');
    check(codec.toLowerCase().contains('hevc'), 'Unexpected video codec');
    check(pixels.contains('10'), 'Native decoder did not expose a 10-bit pixel format');
    final audios = player.state.tracks.audio.where((t) => t.id != 'no' && t.id != 'auto').toList();
    await player.setAudioTrack(audios.last);
    await until(() => player.state.track.audio.id == audios.last.id);
    check(await native.getProperty('aid') == audios.last.id, 'Native audio track did not switch');
    await player.setSubtitleTrack(SubtitleTrack.uri(subtitle, title: 'Synthetic external subtitle'));
    await until(() => player.state.track.subtitle.id != 'no' && player.state.track.subtitle.id != 'auto');
    await player.seek(const Duration(seconds: 5));
    await until(() => player.state.position.inMilliseconds >= 4500);
    await player.setRate(1.5);
    await player.play();
    await until(() => player.state.position.inMilliseconds >= 5400);
    await player.pause();
    final paused = player.state.position.inMilliseconds;
    await player.open(Media(media, httpHeaders: headers, start: const Duration(seconds: 3)), play: false);
    await until(() => player.state.duration.inSeconds >= 7 && player.state.position.inMilliseconds >= 2800 && player.state.position.inMilliseconds < 4000);
    check(authorized > 0 && ranges.isNotEmpty, 'HTTP headers or Range requests not exercised');
    check(errors.isEmpty, 'Native engine emitted playback errors');
    print(jsonEncode({
      'passed': true, 'environment': 'Windows native FFI; headless; software decode',
      'sample': 'synthetic MKV / 3840x2160 / HEVC 10-bit / two AAC audio tracks',
      'engine': await native.getProperty('mpv-version'), 'videoCodec': codec,
      'decodedPixelFormat': pixels,
      'audioSwitch': true, 'externalSubtitleLoaded': true, 'seekMs': paused,
      'resumeMs': player.state.position.inMilliseconds, 'speed': 1.5,
      'authorizedRequests': authorized, 'rangeRequests': ranges.length,
      'iphoneTested': false, 'gpuRenderingTested': false, 'hdrTested': false,
    }));
    await player.stop();
  } finally {
    await errorSubscription.cancel();
    await player.dispose();
    await server.close(force: true);
  }
}
